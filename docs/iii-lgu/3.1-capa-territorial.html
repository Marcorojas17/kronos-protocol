🜁 Aplicación

// Resolución de jurisdicción primaria
function jurisdiccionPrimaria(ubicacion) {
  if (ubicacion.pais === 'MX') return 'mexico-toluca';
  if (ubicacion.region === 'EU') return 'eu-eidas';
  return 'internacional-default';
}

// Para el fundador
const ubicacion = { pais: 'MX', ciudad: 'Toluca' };
const primaria = jurisdiccionPrimaria(ubicacion);
// → 'mexico-toluca'
