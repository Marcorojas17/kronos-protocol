## 🜂 Concepto

## 🜄 Implementación

```javascript
// Persistencia multiplanar
export const PLANOS = ['local', 'indexeddb', 'ipfs', 'arweave', 'ethereum', 'bitcoin', 'shamir'];

export async function persistirMultiplanar(capsula) {
  const resultados = await Promise.allSettled(
    PLANOS.map(async (plano) => {
      const resultado = await persistirEn(plano, capsula);
      return { plano, estado: resultado.ok, hash: capsula.hash };
    })
  );

  return {
    ok: true,
    capsula_id: capsula.id,
    planos: resultados.map(r => r.value),
    regla: 'Local primero. Nube como testigo. Blockchain como notario.',
    veredicto: 'PERSISTENCIA MULTIPLANAR COMPLETA',
    sellado: '◯_● · 51/49/100',
  };
}
```
