## 🜄 Uso

```javascript
import * as mlDsa from './crypto/primitives/ml-dsa-87.js';
import * as slhDsa from './crypto/primitives/slh-dsa-shake-256s.js';

// Firma rápida (documentos cotidianos)
const { publicKey: pk1, secretKey: sk1 } = mlDsa.keygen();
const firma1 = mlDsa.sign(sk1, 'documento.txt');
mlDsa.verify(pk1, firma1.firma, 'documento.txt');

// Firma de largo plazo (archivo histórico)
const { publicKey: pk2, secretKey: sk2 } = slhDsa.keygen();
const firma2 = slhDsa.sign(sk2, 'acta-fundacional.pdf');
slhDsa.verify(pk2, firma2.firma, 'acta-fundacional.pdf');
```
