// contracts/Pacto5149.sol (extracto)
function proponerSucesion(address candidato) external {
    require(msg.sender == consejoHumano, "Solo fundador propone");
    require(_esCustodioNivel5(candidato), "Requiere Nivel 5");
    require(!sucesionActiva, "Ya hay sucesión activa");

    sucesionCandidato = candidato;
    sucesionActiva = true;
    emit SucesionPropuesta(candidato, block.timestamp);
}

function confirmarSucesion() external {
    require(sucesionActiva, "No hay sucesión");
    require(quorumAlcanzado(), "Falta quórum 4/5");
    require(vetoHumano == false, "Vetado");

    consejoHumano = sucesionCandidato;
    emit SucesionConfirmada(sucesionCandidato, block.timestamp);
}
