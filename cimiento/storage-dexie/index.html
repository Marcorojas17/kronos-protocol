v2 · con migraciones 
// Definición del esquema versionado
db.version(1).stores({
  registros: '++id, tipo, timestamp, hash',
  meta: 'clave'
});

db.version(2).stores({
  registros: '++id, tipo, timestamp, hash, estado',
  meta: 'clave',
  migraciones: '++id, fecha, version'
}).upgrade(tx => {
  return tx.table('registros').toCollection()
    .modify(r => { r.estado = r.estado || 'activo'; });
});
