API Pública del Módulo

Reutilizable en todo el ecosistema 
// Importar en cualquier módulo superior
import { CriptoCore } from '../cimiento/cripto-core/core.js';

const core = new CriptoCore();

// 1. Inicializar con contraseña maestra
await core.init('mi-contraseña-maestra');

// 2. Cifrar + firmar + persistir
await core.guardar({ tipo: 'génesis', payload: { ... } });

// 3. Recuperar todo descifrado
const registros = await core.recuperarTodo();

// 4. Verificar integridad de la cadena
const ok = await core.verificarCadena();
